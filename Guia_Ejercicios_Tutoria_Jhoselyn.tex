% --- EJERCICIO 1 ---
\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicio Jh 01 (Guía Quiz 4 --- Pág. 6) · Parábola Cuadrática Completa}
\textbf{Enunciado:} Realice el análisis completo y grafique la función:
$$f(x) = -\frac{1}{2}x^2 + \frac{3}{2}x + \frac{27}{8}$$

\textbf{Desarrollo Paso a Paso de Examen:}
\begin{enumerate}
    \item \textbf{Dominio:} Al ser polinómica de grado 2: $\mathbf{\operatorname{Dom}(f) = \mathbb{R}}$.
    \item \textbf{Cortes con los Ejes Coordenados:}
    \begin{itemize}
        \item Eje $y$ ($x = 0$): $f(0) = \frac{27}{8} = 3.375 \implies \mathbf{P_y = \left(0, \frac{27}{8}\right)}$.
        \item Eje $x$ ($f(x) = 0$): Multiplicando por $-8$: $4x^2 - 12x - 27 = 0 \iff (2x - 9)(2x + 3) = 0$.
        $$\mathbf{x_1 = \frac{9}{2} = 4.5}, \qquad \mathbf{x_2 = -\frac{3}{2} = -1.5}$$
    \end{itemize}
    \item \textbf{Análisis de Paridad:} $f(-x) = -\frac{1}{2}(-x)^2 + \frac{3}{2}(-x) + \frac{27}{8} = -\frac{1}{2}x^2 - \frac{3}{2}x + \frac{27}{8} \neq \pm f(x) \implies$ \textbf{Sin simetría elemental}.
    \item \textbf{Tabla de Signos:} $f(x) = -\frac{1}{2}\left(x + \frac{3}{2}\right)\left(x - \frac{9}{2}\right)$:
    \begin{itemize}
        \item En $]-\infty, -1.5[$: $f(x) < 0$ (Negativa, curva debajo del eje $x$).
        \item En $]-1.5, 4.5[$: $f(x) > 0$ (Positiva, curva sobre el eje $x$).
        \item En $]4.5, +\infty[$: $f(x) < 0$ (Negativa, curva debajo del eje $x$).
    \end{itemize}
    \item \textbf{Asíntotas:} Al ser un polinomio, \textbf{no posee asíntotas} verticales ni horizontales.
    \item \textbf{Vértice y Rango:} $x_v = -\frac{b}{2a} = -\frac{3/2}{2(-1/2)} = \frac{3}{2}$.
    $$y_v = f\left(\frac{3}{2}\right) = -\frac{1}{2}\left(\frac{9}{4}\right) + \frac{3}{2}\left(\frac{3}{2}\right) + \frac{27}{8} = \frac{36}{8} = \frac{9}{2} = 4.5$$
    Como $a = -1/2 < 0$, abre hacia abajo $\implies$ **Máximo Absoluto** en $V\left(\frac{3}{2}, \frac{9}{2}\right)$.
    $$\mathbf{\operatorname{Ran}(f) = \left]-\infty, \frac{9}{2}\right]}$$
\end{enumerate}
\end{ejerciciobox}

% --- EJERCICIO 2 ---
\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicio Jh 02 (Guía Quiz 4 --- Pág. 8) · Polinomio Cúbico Impar}
\textbf{Enunciado:} Analice y grafique: $f(x) = \frac{1}{2}x^3 - \frac{3}{2}x$. (Datos dados en guía: Máx. local en $(-1, 1)$, Mín. local en $(1, -1)$).

\textbf{Solución:}
\begin{enumerate}
    \item \textbf{Dominio:} $\operatorname{Dom}(f) = \mathbb{R}$.
    \item \textbf{Paridad:} $f(-x) = \frac{1}{2}(-x)^3 - \frac{3}{2}(-x) = -\frac{1}{2}x^3 + \frac{3}{2}x = -f(x) \implies$ \textbf{Función IMPAR}.
    Geométricamente presenta \textbf{simetría central respecto al origen $(0, 0)$}.
    \item \textbf{Cortes con el Eje $x$:} $\frac{1}{2}x(x^2 - 3) = 0 \implies \mathbf{x = 0, \quad x = \sqrt{3} \approx 1.732, \quad x = -\sqrt{3} \approx -1.732}$.
    \item \textbf{Tabla de Signos:} Factores: $x, (x-\sqrt{3}), (x+\sqrt{3})$.
    $]-\infty, -\sqrt{3}[$ ($-$); $]-\sqrt{3}, 0[$ ($+$); $]0, \sqrt{3}[$ ($-$); $]\sqrt{3}, +\infty[$ ($+$).
    \item \textbf{Monotonía y Rango:} Creciente en $]-\infty, -1] \cup [1, +\infty[$; Decreciente en $[-1, 1]$.
    $$\mathbf{\operatorname{Ran}(f) = \mathbb{R}}$$
\end{enumerate}
\end{ejerciciobox}

% --- EJERCICIO 3 ---
\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicio Jh 03 (Guía Quiz 4 --- Pág. 10) · Polinomio Bicuadrático Par}
\textbf{Enunciado:} Analice y grafique: $f(x) = \frac{1}{4}x^4 - \frac{3}{2}x^2 + \frac{5}{4}$. (Datos: Máx. en $(0, 1.25)$, Mínimos absolutos en $(\pm\sqrt{3}, -1)$).

\textbf{Solución:}
\begin{enumerate}
    \item \textbf{Dominio:} $\operatorname{Dom}(f) = \mathbb{R}$.
    \item \textbf{Paridad:} Exponentes pares $\implies f(-x) = f(x) \implies$ \textbf{Función PAR (Simetría axial eje $y$)}.
    \item \textbf{Cortes con el Eje $x$:} Factorizando como trinomio: $\frac{1}{4}(x^4 - 6x^2 + 5) = 0 \iff (x^2 - 1)(x^2 - 5) = 0$.
    $$\mathbf{x = \pm 1}, \qquad \mathbf{x = \pm\sqrt{5} \approx \pm 2.236}$$
    \item \textbf{Extremos y Rango:} Mínimo absoluto en $y = -1$. Como las ramas crecen hacia $+\infty$:
    $$\mathbf{\operatorname{Ran}(f) = [-1, +\infty[}$$
\end{enumerate}
\end{ejerciciobox}

% --- EJERCICIO 4 ---
\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicio Jh 04 (Guía Quiz 4 --- Pág. 12 --- ¡PREGUNTA ESTRELLA DE QUIZ!) · Racional 3 Ramas}
\textbf{Enunciado:} Realice el análisis formal de 7 pasos y bosqueje la gráfica cartesiana de:
$$f(x) = \frac{x^2 - 1}{x^2 - 4}$$
\textit{Dato provisto en la guía: la rama central tiene un máximo local en $\left(0, \frac{1}{4}\right)$.}

\textbf{Desarrollo Riguroso de Examen:}
\begin{enumerate}
    \item \textbf{Dominio:} El denominador no puede anularse: $x^2 - 4 = 0 \iff x = \pm 2$.
    $$\mathbf{\operatorname{Dom}(f) = \mathbb{R} \setminus \{-2, 2\} = ]-\infty, -2[ \cup ]-2, 2[ \cup ]2, +\infty[}$$
    Divide el plano cartesiano en 3 regiones disjuntas.
    \item \textbf{Intersecciones con los Ejes:}
    \begin{itemize}
        \item Eje $y$ ($x = 0$): $f(0) = \frac{0 - 1}{0 - 4} = \frac{1}{4} \implies \mathbf{P_y = \left(0, \frac{1}{4}\right)}$.
        \item Eje $x$ ($f(x) = 0$): $x^2 - 1 = 0 \implies \mathbf{x = 1, \; x = -1} \implies \mathbf{P_{x_1}(1, 0), \; P_{x_2}(-1, 0)}$.
    \end{itemize}
    \item \textbf{Paridad:} $f(-x) = \frac{(-x)^2 - 1}{(-x)^2 - 4} = \frac{x^2 - 1}{x^2 - 4} = f(x) \implies$ \textbf{Función PAR} (simetría axial respecto al eje $y$).
    \item \textbf{Tabla de Signos por Factores:} $f(x) = \frac{(x-1)(x+1)}{(x-2)(x+2)}$ con puntos críticos $-2, -1, 1, 2$:
    \begin{center}
    \begin{tabular}{lccccc}
    \toprule
    \textbf{Factor} & $]-\infty, -2[$ & $]-2, -1[$ & $]-1, 1[$ & $]1, 2[$ & $]2, +\infty[$ \\
    \midrule
    $x+2$ & $-$ & $+$ & $+$ & $+$ & $+$ \\
    $x+1$ & $-$ & $-$ & $+$ & $+$ & $+$ \\
    $x-1$ & $-$ & $-$ & $-$ & $+$ & $+$ \\
    $x-2$ & $-$ & $-$ & $-$ & $-$ & $+$ \\
    \midrule
    \textbf{Signo $f(x)$} & \textbf{$+$} & \textbf{$-$} & \textbf{$+$} & \textbf{$-$} & \textbf{$+$} \\
    \bottomrule
    \end{tabular}
    \end{center}
    \item \textbf{Asíntotas:}
    \begin{itemize}
        \item \textbf{Verticales:} En ceros del denominador no cancelados: $\mathbf{x = -2}$ y $\mathbf{x = 2}$.
        \item \textbf{Horizontal:} Grado numerador = grado denominador ($2 = 2$): $\lim_{x \to \pm\infty} \frac{x^2-1}{x^2-4} = \frac{1}{1} = 1 \implies \mathbf{y = 1}$.
        \item \textbf{Oblicua:} No existe, pues $\operatorname{grad}(P) = \operatorname{grad}(Q)$.
    \end{itemize}
    \item \textbf{Extremos y Rango:} La rama central está acotada superiormente por su máximo local en $y = \frac{1}{4}$, y cae a $-\infty$ al aproximarse a $x \to \pm 2$. Las ramas laterales son asintóticas a $y = 1$ por arriba y crecen a $+\infty$:
    $$\mathbf{\operatorname{Ran}(f) = \left]-\infty, \frac{1}{4}\right] \cup ]1, +\infty[}$$
\end{enumerate}
\end{ejerciciobox}

% --- EJERCICIO 5 ---
\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicio Jh 05 (Guía Quiz 4 --- Pág. 14) · Función Racional con Asíntota Oblicua}
\textbf{Enunciado:} Analice y grafique: $f(x) = \frac{x^3 - 4x}{x^2 - 1}$. (\textit{Dato: monótona creciente en cada intervalo}).

\textbf{Solución:}
\begin{enumerate}
    \item \textbf{Dominio:} $x^2 - 1 \neq 0 \implies \mathbf{\operatorname{Dom}(f) = \mathbb{R} \setminus \{-1, 1\}}$.
    \item \textbf{Paridad:} $f(-x) = \frac{(-x)^3 - 4(-x)}{(-x)^2 - 1} = \frac{-x^3 + 4x}{x^2 - 1} = -f(x) \implies$ \textbf{Función IMPAR} (simetría al origen).
    \item \textbf{Cortes:} Eje $x$: $x(x - 2)(x + 2) = 0 \implies \mathbf{x = -2, \quad x = 0, \quad x = 2}$.
    \item \textbf{Asíntotas Verticales:} $\mathbf{x = -1}$ y $\mathbf{x = 1}$.
    \item \textbf{Asíntota Oblicua (División Polinomial):}
    $$x^3 - 4x = x(x^2 - 1) - 3x \implies f(x) = x - \frac{3x}{x^2 - 1}$$
    Como $\lim_{x \to \pm\infty} \left(-\frac{3x}{x^2 - 1}\right) = 0$, la recta asíntota oblicua es: $\mathbf{y = x}$.
    \item \textbf{Rango:} $\mathbf{\operatorname{Ran}(f) = \mathbb{R}}$.
\end{enumerate}
\end{ejerciciobox}

% --- EJERCICIO 6 ---
\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicio Jh 06 (Guía Quiz 4 --- Pág. 16) · Racional con Rango por Extremos}
\textbf{Enunciado:} Analice y grafique: $f(x) = \frac{x^2 - 9}{x^2 - 1}$. (\textit{Dato: la rama central tiene un mínimo en $(0, 9)$}).

\textbf{Solución:}
\begin{enumerate}
    \item \textbf{Dominio:} $\operatorname{Dom}(f) = \mathbb{R} \setminus \{-1, 1\}$.
    \item \textbf{Paridad:} $f(-x) = f(x) \implies$ \textbf{Función PAR}.
    \item \textbf{Cortes:} Eje $x$: $x^2 - 9 = 0 \implies \mathbf{x = \pm 3}$. Eje $y$: $f(0) = \frac{-9}{-1} = \mathbf{9}$.
    \item \textbf{Asíntotas:} Verticales en $\mathbf{x = \pm 1}$; Horizontal en $\mathbf{y = 1}$.
    \item \textbf{Rango:} La rama central tiene un mínimo en $y = 9$ y sube a $+\infty$. Las laterales viven debajo de $y = 1$:
    $$\mathbf{\operatorname{Ran}(f) = ]-\infty, 1[ \cup [9, +\infty[}$$
\end{enumerate}
\end{ejerciciobox}

% --- EJERCICIO 7 ---
\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicio Jh 07 (Guía Quiz 4 --- Pág. 18) · Función con Raíz Cuadrada}
\textbf{Enunciado:} Analice y grafique: $f(x) = \sqrt{\frac{x^2 - 1}{x^2 - 4}}$. (\textit{Dato: máximo de rama central en $\left(0, \frac{1}{2}\right)$}).

\textbf{Solución:}
\begin{enumerate}
    \item \textbf{Dominio:} Radicando no negativo con denominador no nulo: $\frac{x^2 - 1}{x^2 - 4} \ge 0$ y $x \neq \pm 2$.
    Usando la tabla de signos del Ejercicio 4, la fracción es $\ge 0$ en:
    $$\mathbf{\operatorname{Dom}(f) = ]-\infty, -2[ \cup [-1, 1] \cup ]2, +\infty[}$$
    \item \textbf{Paridad:} $f(-x) = f(x) \implies$ \textbf{Función PAR}.
    \item \textbf{Cortes:} Eje $x$: $(\pm 1, 0)$. Eje $y$: $\left(0, \sqrt{1/4}\right) = \left(0, \frac{1}{2}\right)$.
    \item \textbf{Asíntotas:} Verticales en $\mathbf{x = \pm 2}$; Horizontal en $\mathbf{y = \sqrt{1} = 1}$.
    \item \textbf{Rango:} Al ser raíz cuadrada principal, $f(x) \ge 0$:
    $$\mathbf{\operatorname{Ran}(f) = [0, 1/2] \cup ]1, +\infty[}$$
\end{enumerate}
\end{ejerciciobox}

\section*{\color{ponchoRed}PARTE 2: Inyectividad, Sobreyectividad y Función Inversa (Guía 2.6)}

\begin{metodobox}{\textbf{Plantilla} La Plantilla Oficial para Demostrar Inyectividad}
Para obtener la máxima puntuación en demostraciones de inyectividad, escribe siempre estos 3 pasos:
\begin{enumerate}
    \item \textit{"Sean $x_1, x_2 \in \operatorname{Dom}(f)$ tales que $f(x_1) = f(x_2)$."}
    \item \textit{Desarrollo algebraico hasta cancelar términos idénticos.}
    \item \textit{Conclusión: "Por tanto, $x_1 = x_2$, lo que demuestra formalmente que $f$ es inyectiva."}
\end{enumerate}
\end{metodobox}

\begin{trampa}
\textbf{¡El error más común con raíces cuadradas!}\\
Al despejar $x$ en expresiones cuadráticas, \textbf{nunca} omitas el valor absoluto. La regla matemática estricta es:
$$\sqrt{A^2} = |A|$$
Primero escribe $|x - h| = \sqrt{\dots}$ y luego justifica por qué el signo resultante es $+$ o $-$ basándote en el intervalo de dominio dado en la hipótesis del problema.
\end{trampa}

% --- EJERCICIO 8 ---
\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicio Jh 08 (Guía 2.6 --- Pág. 15) · Parábola Biyectiva y Función Inversa}
\textbf{Enunciado:} Considere la función $f: [0, +\infty[ \to [1, +\infty[$ definida por $f(x) = x^2 + 1$. Demuestre que es biyectiva y determine su función inversa $f^{-1}(x)$.

\textbf{Demostración Formal de Examen:}
\begin{enumerate}
    \item \textbf{Demostración de Inyectividad:}
    Sean $x_1, x_2 \in [0, +\infty[$ tales que $f(x_1) = f(x_2)$:
    $$x_1^2 + 1 = x_2^2 + 1 \implies x_1^2 = x_2^2 \implies \sqrt{x_1^2} = \sqrt{x_2^2} \implies |x_1| = |x_2|$$
    Como la hipótesis establece que $x_1 \ge 0$ y $x_2 \ge 0$, se tiene que $|x_1| = x_1$ y $|x_2| = x_2$:
    $$\mathbf{x_1 = x_2} \implies \mathbf{f \text{ es estrictamente INYECTIVA.}}$$
    \item \textbf{Demostración de Sobreyectividad:}
    Sea $y \in [1, +\infty[$ un elemento cualquiera del codominio. Buscamos $x \in [0, +\infty[$ tal que $y = f(x)$:
    $$y = x^2 + 1 \implies x^2 = y - 1 \implies |x| = \sqrt{y - 1}$$
    Como $x \ge 0$, $|x| = x$. Además, como $y \ge 1 \implies y - 1 \ge 0$, la raíz existe en $\mathbb{R}$:
    $$x = +\sqrt{y - 1} \in [0, +\infty[$$
    Todo elemento del codominio tiene al menos una preimagen real $\implies \mathbf{f \text{ es SOBREYECTIVA.}}$
    \item \textbf{Cálculo de la Función Inversa:}
    Al ser inyectiva y sobreyectiva, $f$ es **BIYECTIVA**, por lo que su inversa existe de forma única:
    $$\mathbf{f^{-1}: [1, +\infty[ \longrightarrow [0, +\infty[, \qquad f^{-1}(x) = \sqrt{x - 1}}$$
    \item \textbf{Propiedad Gráfica:} Las curvas de $f(x)$ y $f^{-1}(x)$ son **simétricas respecto a la recta diagonal $y = x$**.
\end{enumerate}
\end{ejerciciobox}

\section*{\color{ponchoRed}PARTE 3: Transformaciones de Funciones y Formas Canónicas (Guía 2.5)}

\begin{metodobox}{\textbf{Regla} Fórmula Maestra de Transformación y Reglas de Signos}
$$g(x) = a \cdot f(b(x - h)) + k$$
\begin{itemize}
    \item \textbf{Desplazamiento Horizontal ($h$):} $(x - h)$ desplaza $h$ unidades a la \textbf{derecha}; $(x + h)$ desplaza a la \textbf{izquierda}.
    \item \textbf{Desplazamiento Vertical ($k$):} $+k$ desplaza hacia \textbf{arriba}; $-k$ desplaza hacia \textbf{abajo}.
    \item \textbf{Escala Vertical ($a$):} $|a| > 1$ dilata; $|a| < 1$ comprime. Si $a < 0$: \textbf{reflexión en el eje $X$}.
    \item \textbf{Escala Horizontal ($b$):} $|b| > 1$ \textbf{comprime} por factor $\frac{1}{|b|}$; $|b| < 1$ \textbf{dilata}. Si $b < 0$: \textbf{reflexión en el eje $Y$}.
\end{itemize}
\end{metodobox}

% --- EJERCICIO 9 ---
\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicio Jh 09 (Guía 2.5 --- Pág. 1) · Forma Canónica y Transformaciones Cuadráticas}
\textbf{Enunciado:} A partir de $f(x) = x^2$, determine todas las transformaciones necesarias para obtener:
$$g(x) = -\frac{3}{2}(2x + 4)^2 + 5$$
Exprese la función en la forma estándar $a(x - h)^2 + k$ y determine las coordenadas de su vértice.

\textbf{Solución Paso a Paso:}
\begin{enumerate}
    \item \textbf{Factorización Interna Obligatoria:} El coeficiente de $x$ dentro del binomio debe ser $1$:
    $$2x + 4 = 2(x + 2) \implies (2x + 4)^2 = [2(x + 2)]^2 = 2^2 \cdot (x + 2)^2 = 4(x + 2)^2$$
    \item \textbf{Obtención de la Forma Canónica:}
    $$g(x) = -\frac{3}{2} \cdot \left[4(x + 2)^2\right] + 5 = -6(x + 2)^2 + 5$$
    \item \textbf{Parámetros y Vértice:}
    $$\mathbf{a = -6, \quad h = -2, \quad k = 5} \implies \mathbf{V(h, k) = (-2, 5)}$$
    Como $a = -6 < 0$, la parábola abre hacia abajo $\implies$ Máximo absoluto en $y = 5$, con $\mathbf{\operatorname{Ran}(g) = (-\infty, 5]}$.
    \item \textbf{Secuencia Ordenada de Transformaciones desde $f(x) = x^2$:}
    \begin{enumerate}
        \item Traslación horizontal de $2$ unidades hacia la izquierda: $(x + 2)^2$.
        \item Dilatación vertical de factor $6$: $6(x + 2)^2$.
        \item Reflexión respecto al eje $X$: $-6(x + 2)^2$.
        \item Traslación vertical de $5$ unidades hacia arriba: $-6(x + 2)^2 + 5$.
    \end{enumerate}
    \item \textbf{Equivalencia con la Interpretación Directa de la Guía Oficial:}
    La guía oficial también valida: $2\text{ u a la izquierda}$, compresión horizontal de factor $\frac{1}{2}$, dilatación vertical de factor $\frac{3}{2}$, reflexión en eje $X$, y $5\text{ u hacia arriba}$. Ambas son equivalentes pues $(2x)^2 = 4x^2$ y $4 \times \frac{3}{2} = 6$.
\end{enumerate}
\end{ejerciciobox}

\begin{ejerciciobox}{\textbf{Ejercicio} Ejercicios de Reconstrucción de Funciones (Guía 2.5 --- Págs. 2 a 4)}
\begin{enumerate}
    \item \textbf{Reconstrucción de Valor Absoluto:} $f(x) = |x|$, vértice $(-3, 4)$, pasa por $(-1, 1)$, abre hacia abajo:
    \begin{align*}
    g(x) &= a|x + 3| + 4 \implies 1 = a|-1 + 3| + 4 \implies 1 = 2a + 4\\
    &\implies a = -\tfrac{3}{2} \implies \mathbf{g(x) = -\frac{3}{2}|x + 3| + 4}
    \end{align*}
    \item \textbf{Reconstrucción de Radical:} $f(x) = \sqrt{x}$, inicio $(5, -2)$, reflejada en eje $X$, pasa por $(9, -6)$:
    \begin{align*}
    g(x) &= -a\sqrt{x - 5} - 2 \implies -6 = -a\sqrt{9 - 5} - 2 \implies -4 = -2a\\
    &\implies a = 2 \implies \mathbf{g(x) = -2\sqrt{x - 5} - 2}
    \end{align*}
    \item \textbf{Reconstrucción de Parábola:} $f(x) = x^2$, vértice $V(4, -3)$, pasa por $(6, 5)$:
    \begin{align*}
    g(x) &= a(x - 4)^2 - 3 \implies 5 = a(6 - 4)^2 - 3 \implies 8 = 4a\\
    &\implies a = 2 \implies \mathbf{g(x) = 2(x - 4)^2 - 3}
    \end{align*}
\end{enumerate}
