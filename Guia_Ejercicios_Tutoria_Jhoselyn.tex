\ejerciciopage{Ejercicio Jh 01 \textbar\ Parábola Cuadrática, Vértice y Cortes}{Quiz 4 $\cdot$ Pág. 1}{7 Pasos}

\begin{enunciadobox}
\textbf{\color{amarunTeal}ENUNCIADO:} Analice completamente y determine el vértice canónico, cortes axiales, simetría y rango de la función cuadrática:
\[ f(x) = -\frac{1}{2}x^2 + \frac{3}{2}x + \frac{27}{8} \]
\end{enunciadobox}

\begin{solucionbox}
\textbf{\color{ponchoRed}SOLUCIÓN DEDUCTIVA PASO A PASO:}
\begin{enumerate}
    \item \textbf{Vértice Canónico $V(h, k)$:}
    \begin{align*}
        h &= -\frac{b}{2a} = -\frac{3/2}{2(-1/2)} = \frac{3}{2} = 1.5 \\
        k &= f(1.5) = -\frac{1}{2}\left(\frac{3}{2}\right)^2 + \frac{3}{2}\left(\frac{3}{2}\right) + \frac{27}{8} \\
        &= -\frac{9}{8} + \frac{18}{8} + \frac{27}{8} = \frac{36}{8} = \frac{9}{2} = 4.5
    \end{align*}
    El vértice canónico es $\mathbf{V\left(\frac{3}{2}, \frac{9}{2}\right)}$ con eje de simetría vertical en $x = \frac{3}{2}$.
    \item \textbf{Cortes Axiales Exactos:}
    \begin{align*}
        \text{Corte } Y (x = 0): \quad f(0) &= \frac{27}{8} = 3.375 \implies \mathbf{\left(0, \frac{27}{8}\right)} \\
        \text{Cortes } X (f(x) = 0): \quad x^2 - 3x - \frac{27}{4} &= 0 \implies \left(x - \frac{9}{2}\right)\left(x + \frac{3}{2}\right) = 0 \\
        x_1 = \frac{9}{2} = 4.5, \quad x_2 &= -\frac{3}{2} = -1.5 \implies \mathbf{\left(\frac{9}{2}, 0\right)}, \; \mathbf{\left(-\frac{3}{2}, 0\right)}
    \end{align*}
\end{enumerate}

\ejerciciopage{Ejercicio Jh 02 \textbar\ Función Cúbica, Simetría al Origen y Extremos}{Quiz 4 $\cdot$ Pág. 1}{Simetría Impar}

\begin{enunciadobox}
\textbf{\color{amarunTeal}ENUNCIADO:} Demuestre analíticamente la paridad, calcule los ceros reales y determine los extremos locales de la función cúbica:
\[ f(x) = \frac{1}{2}x^3 - \frac{3}{2}x \]
\end{enunciadobox}

\begin{solucionbox}
\textbf{\color{ponchoRed}SOLUCIÓN DEDUCTIVA PASO A PASO:}
\begin{enumerate}
    \item \textbf{Simetría Analítica (Paridad):}
    \begin{align*}
        f(-x) &= \frac{1}{2}(-x)^3 - \frac{3}{2}(-x) \\
        &= -\frac{1}{2}x^3 + \frac{3}{2}x \\
        &= -\left(\frac{1}{2}x^3 - \frac{3}{2}x\right) = -f(x) \quad \implies \text{\textbf{Función Impar}}
    \end{align*}
    Posee simetría central respecto al origen de coordenadas $(0, 0)$.
    \item \textbf{Cortes Axiales y Puntos Críticos:}
    \begin{align*}
        \frac{1}{2}x^3 - \frac{3}{2}x &= 0 \implies \frac{1}{2}x\left(x^2 - 3\right) = 0 \\
        x = 0, \quad x &= \pm\sqrt{3} \approx \pm 1.732 \implies \mathbf{(0, 0)}, \; \mathbf{(-\sqrt{3}, 0)}, \; \mathbf{(\sqrt{3}, 0)} \\
        \text{Extremos Locales:} \quad f(1) &= \frac{1}{2} - \frac{3}{2} = -1 \implies \mathbf{(1, -1)} \text{ (Mínimo local)} \\
        f(-1) &= -f(1) = 1 \implies \mathbf{(-1, 1)} \text{ (Máximo local)}
    \end{align*}
\end{enumerate}

\ejerciciopage{Ejercicio Jh 03 \textbar\ Polinomio Bicuadrático y Mínimo Absoluto}{Quiz 4 $\cdot$ Pág. 1}{Simetría Par}

\begin{enunciadobox}
\textbf{\color{amarunTeal}ENUNCIADO:} Halle las cuatro raíces reales, determine la simetría axial y deduzca algebraicamente el rango de:
\[ f(x) = \frac{1}{4}x^4 - \frac{3}{2}x^2 + \frac{5}{4} \]
\end{enunciadobox}

\begin{solucionbox}
\textbf{\color{ponchoRed}SOLUCIÓN DEDUCTIVA PASO A PASO:}
\begin{enumerate}
    \item \textbf{Simetría Par y Factorización de Raíces:}
    \begin{align*}
        f(-x) &= \frac{1}{4}(-x)^4 - \frac{3}{2}(-x)^2 + \frac{5}{4} = f(x) \quad \implies \text{\textbf{Función Par (Eje Y)}} \\
        f(x) &= \frac{1}{4}\left(x^4 - 6x^2 + 5\right) = \frac{1}{4}\left(x^2 - 1\right)\left(x^2 - 5\right) = 0 \\
        x &= \pm 1, \quad x = \pm\sqrt{5} \approx \pm 2.236 \implies \mathbf{(\pm 1, 0)}, \; \mathbf{(\pm\sqrt{5}, 0)}
    \end{align*}
    \item \textbf{Mínimo Absoluto por Completación de Cuadrados:}
    \begin{align*}
        f(x) &= \frac{1}{4}\left[(x^2 - 3)^2 - 9 + 5\right] \\
        &= \frac{1}{4}\left(x^2 - 3\right)^2 - 1 \\
        \implies \min f(x) &= -1 \quad \text{cuando } x^2 = 3 \iff x = \pm\sqrt{3} \approx \pm 1.732
    \end{align*}
\end{enumerate}

\ejerciciopage{Ejercicio Jh 04 \textbar\ Función Racional de Tres Ramas y Asíntotas}{Quiz 4 $\cdot$ Pág. 2}{Asíntotas y Rango}

\begin{enunciadobox}
\textbf{\color{amarunTeal}ENUNCIADO:} Desarrolle el análisis de asíntotas, simetría y determine algebraicamente el rango de la función racional:
\[ f(x) = \frac{x^2 - 1}{x^2 - 4} \]
\end{enunciadobox}

\begin{solucionbox}
\textbf{\color{ponchoRed}SOLUCIÓN DEDUCTIVA PASO A PASO:}
\begin{enumerate}
    \item \textbf{Asíntotas Verticales y Horizontal:}
    \begin{align*}
        \text{Verticales (Polos):} \quad x^2 - 4 &= 0 \implies \mathbf{x = -2}, \quad \mathbf{x = 2} \\
        \text{Horizontal:} \quad y &= \lim_{x \to \pm\infty} \frac{x^2 - 1}{x^2 - 4} = 1 \implies \mathbf{y = 1}
    \end{align*}
    \item \textbf{Deducción del Rango por Despeje:}
    \begin{align*}
        y &= \frac{x^2 - 1}{x^2 - 4} \implies y(x^2 - 4) = x^2 - 1 \\
        x^2(y - 1) &= 4y - 1 \implies x^2 = \frac{4y - 1}{y - 1} \ge 0 \\[3pt]
        \implies \operatorname{Ran}(f) &= \mathbf{\left(-\infty, \frac{1}{4}\right] \cup (1, +\infty)}
    \end{align*}
\end{enumerate}

\ejerciciopage{Ejercicio Jh 05 \textbar\ Racional Impar con Asíntota Oblicua}{Quiz 4 $\cdot$ Pág. 2}{División Polinomial}

\begin{enunciadobox}
\textbf{\color{amarunTeal}ENUNCIADO:} Halle las asíntotas completas (incluyendo oblicua), simetría y cortes de la función racional:
\[ f(x) = \frac{x^3 - 4x}{x^2 - 1} \]
\end{enunciadobox}

\begin{solucionbox}
\textbf{\color{ponchoRed}SOLUCIÓN DEDUCTIVA PASO A PASO:}
\begin{enumerate}
    \item \textbf{Asíntota Oblicua por División Polinomial:}
    \begin{align*}
        \frac{x^3 - 4x}{x^2 - 1} &= \frac{x(x^2 - 1) - 3x}{x^2 - 1} = x - \frac{3x}{x^2 - 1} \\
